        $(document).ready(function() {
            console.log('✅ Test JS cargado');

            // Botón para abrir modal
            $('#openModalBtn').on('click', function() {
                console.log('Click en ABRIR MODAL');
                $('#testModal').modal('show');
            });

            // Botones de rol
            $('.role-btn').on('click', function() {
                var role = $(this).data('role');
                console.log('Rol seleccionado:', role);
                
                if (role === 'medico' || role === 'enfermera') {
                    // Abrir modal automáticamente
                    $('#testModal').modal('show');
                    // Actualizar título
                    $('.modal-title').text('DATOS PROFESIONALES - ' + role.toUpperCase());
                }
            });

            // Guardar datos
            $('#saveTestBtn').on('click', function() {
                var specialty = $('#testSpecialty').val();
                var license = $('#testLicense').val();
                console.log('Especialidad:', specialty);
                console.log('Licencia:', license);
                alert('Datos guardados: ' + specialty + ' - ' + license);
                $('#testModal').modal('hide');
            });
        });
